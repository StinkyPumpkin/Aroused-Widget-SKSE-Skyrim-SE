cmake_minimum_required(VERSION 3.21)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "")
# Renamed 2026-08-11: ArousedWidgetClaude -> ArousedWidget for Nexus release.
# DLL name and SKSE-registered plugin name both follow the project name.
# iHUDClaude's ArousedWidgetBridge kReceiver must match this name.
# 3.8.0 (2026-09-27): version follows the Nexus page (3.7 -> 3.8); the repo was at 0.4.0.
project(ArousedWidget VERSION 3.8.0 LANGUAGES CXX)

# Output to MO2 mods folder
if(DEFINED ENV{SKYRIM_MODS_FOLDER} AND IS_DIRECTORY "$ENV{SKYRIM_MODS_FOLDER}")
    set(OUTPUT_FOLDER "$ENV{SKYRIM_MODS_FOLDER}/Aroused Widget--Claude")
endif()

# 3.8.0: CommonLibSSE-NG from alandtse (github.com/alandtse/CommonLibSSE-NG, formerly CommonLibVR, branch `ng`),
# built from source out of the pinned submodule extern/CommonLibSSE-NG (tag v9.1.0). This is the lineage that reads
# the Address Library v5 database and sets the AddressLibraryV5 flag, so 1.7.x SKSE loads the DLL. See BUILD.md.
set(CLIB_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/extern/CommonLibSSE-NG")
if(NOT EXISTS "${CLIB_SOURCE_DIR}/cmake/CommonLibSSE.cmake")
    message(FATAL_ERROR "extern/CommonLibSSE-NG is empty. Run: git submodule update --init --recursive --depth 1")
endif()
if(NOT EXISTS "${CLIB_SOURCE_DIR}/extern/openvr/lib/win64/openvr_api.lib")
    message(FATAL_ERROR "extern/CommonLibSSE-NG/extern/openvr is empty (VR is enabled). Run: git submodule update --init --recursive --depth 1")
endif()

# Every CACHE FORCE must precede add_subdirectory: option() inside NG cannot override an existing cache entry.
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)                 # NG defaults ON -> find_package(Catch2 REQUIRED)
set(SKSE_SUPPORT_XBYAK OFF CACHE BOOL "" FORCE)          # the widget uses no trampoline
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)   # write_branch diagnostics; the widget has no hooks, and ON
                                                         # FetchContent-downloads minhook at configure time
# Same runtime set as the 0.4.0 DLL (the old vcpkg port built SE+AE+VR): one DLL, runtime dispatch.
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)             # AE = 1.6.x AND 1.7.x (NG classifies minor >= 6 as AE)
set(ENABLE_SKYRIM_VR ON CACHE BOOL "" FORCE)
set(REX_OPTION_INI OFF CACHE BOOL "" FORCE)
set(REX_OPTION_JSON OFF CACHE BOOL "" FORCE)
set(REX_OPTION_TOML OFF CACHE BOOL "" FORCE)

add_subdirectory("${CLIB_SOURCE_DIR}" CommonLibSSE-NG)
# project() inside the subdirectory keeps CommonLibSSE_VERSION in that scope; read the pinned version from its manifest.
file(READ "${CLIB_SOURCE_DIR}/vcpkg.json" _clib_manifest)
string(JSON AW_CLIB_VERSION GET "${_clib_manifest}" "version-semver")
message(STATUS "CommonLibSSE-NG ${AW_CLIB_VERSION} from ${CLIB_SOURCE_DIR}")
# NG's source build does not include its plugin helper (only the prebuilt path does).
include("${CLIB_SOURCE_DIR}/cmake/CommonLibSSE.cmake")

# Declaration generated by add_commonlibsse_plugin (SKSEPlugin_Version + SKSEPlugin_Query):
#   - Address Library independence (versionIndependence = AddressLibraryPostAE), no fixed runtime list
#   - StructCompatibility::Independent = AddressLibraryV5 | NoStructUse in versionIndependenceEx
# MINIMUM_SKSE_VERSION must be a full x.y.z triplet: a bare "0" (the helper's default) generates
# REL::Version{ 0, , , 0 } and fails to compile.
add_commonlibsse_plugin(${PROJECT_NAME} MINIMUM_SKSE_VERSION 0.0.0 SOURCES
    src/plugin.cpp
    src/WidgetController.cpp
    src/ArousalReader.cpp
    src/JsonStore.cpp
    src/Settings.cpp
    src/HudUI.cpp
    src/SettingsUI.cpp
    src/Visibility.cpp
    src/HotkeyHandler.cpp
    src/iHUDBridge.cpp
)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_23)
target_precompile_headers(${PROJECT_NAME} PRIVATE PCH.h)

# Logged at load: "ArousedWidget v<version>" and "built on CommonLibSSE-NG <version>".
target_compile_definitions(${PROJECT_NAME} PRIVATE
    AW_VERSION="${PROJECT_VERSION}"
    AW_COMMONLIB_VERSION="${AW_CLIB_VERSION}"
)

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_SOURCE_DIR}/src
    ${CMAKE_SOURCE_DIR}/include
)

if(DEFINED OUTPUT_FOLDER)
    set(DLL_FOLDER "${OUTPUT_FOLDER}/SKSE/Plugins")
    set(TEX_FOLDER "${OUTPUT_FOLDER}/Interface/HUDWidgets")

    message(STATUS "SKSE plugin output: ${DLL_FOLDER}")
    message(STATUS "Texture root      : ${TEX_FOLDER}")

    add_custom_command(
        TARGET "${PROJECT_NAME}" POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${DLL_FOLDER}"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:${PROJECT_NAME}>" "${DLL_FOLDER}/$<TARGET_FILE_NAME:${PROJECT_NAME}>"
        VERBATIM
    )

    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        add_custom_command(
            TARGET "${PROJECT_NAME}" POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_PDB_FILE:${PROJECT_NAME}>" "${DLL_FOLDER}/$<TARGET_PDB_FILE_NAME:${PROJECT_NAME}>"
            VERBATIM
        )
    endif()
endif()
